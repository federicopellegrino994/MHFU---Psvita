project(prepare_local LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(R "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)

target_include_directories(prepare-local PRIVATE ${R}/launcher/native ${R}/games/mhfu/tools)
target_link_libraries(prepare-local PRIVATE mhf_prx)
set(MHF_BUILD_TESTS OFF)
add_subdirectory(${R}/third_party/psprecomp psprecomp)
add_executable(mhf_recomp ${R}/platforms/psp/recomp/main.cpp ${R}/platforms/psp/recomp/analysis.cpp ${R}/platforms/psp/recomp/emitter.cpp ${R}/platforms/psp/recomp/instruction_emitter.cpp)
